#!/usr/bin/env python3
"""
v4 N型结构三态 (v0.4, MOS 2026-08-14 11:49: 拐点源改用原始收盘分形拐点 W=2)
核心定义 (MOS):
  - 拐点序列: 原始收盘分形拐点(左右W日严格极值, 同型连续保留更极值者), 与v3解耦
  - 相邻比较: 当前低点只与上一个低点比较, 当前高点只与上一个高点比较
  - 上涨N: 低点抬高 且 高点抬高; 下跌N: 低点降低 且 高点降低
  - 剩下都是横盘 (横盘交易不具有显著胜率, 页面橙色背景)
  - 拐点极值无关紧要, 只需标注拐点K线
分段规则:
  任意相邻同型比较违规(含相等) → 在最近锚点处断开(共享拐点)
    上涨结构锚点=H: 低点或高点回落 → 结束于最近H
    下跌结构锚点=L: 低点或高点抬升 → 结束于最近L
    锚点违规: 违规比较的两点都在新结构内 → 直接取方向
    非锚点违规: 违规比较跨边界 → 方向待定(0), 等边界内两个比较同向再确立
    确立方向需N单元不越当前结构起点
验证: 结构内所有相邻同型比较与方向一致(≥4拐点)才是有效N, 否则横盘
跨边界比较一律忽略
已通过: MOS三例+边界例单元测试; 10000随机序列 vs 独立oracle 全部一致
输出: pages/v4-states.json (透传 v3 字段 + v4 state/segments/swings/v4structs)
"""
import json, os, sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from config import PIPELINE_DIR

V3 = os.path.join(PIPELINE_DIR, 'pages', 'v3-states.json')
OUT = os.path.join(PIPELINE_DIR, 'pages', 'v4-states.json')
W = 3  # 分形窗口

def fractal_pivots(closes, W=2):
    """原始收盘分形拐点: 左右W日严格极值; 同型连续保留更极值者"""
    piv = []
    for i in range(W, len(closes) - W):
        c = closes[i]
        if c is None:
            continue
        left = closes[i - W:i]
        right = closes[i + 1:i + W + 1]
        win = list(left) + list(right)
        if all(x is not None and c > x for x in win):
            t = 'H'
        elif all(x is not None and c < x for x in win):
            t = 'L'
        else:
            continue
        if piv and piv[-1]['t'] == t:
            if (t == 'H' and c > piv[-1]['v']) or (t == 'L' and c < piv[-1]['v']):
                piv[-1] = {'i': i, 't': t, 'v': round(c, 2)}
            continue
        piv.append({'i': i, 't': t, 'v': round(c, 2)})
    return piv

def sign(x):
    return 1 if x > 0 else (-1 if x < 0 else 0)

def process(piv, n, dates, last_close=None):
    """拐点序列 → (v4state, v4segments, swings, v4structs)
    last_close: 最新收盘 — 末段无分形极值可取时(MOS 12:22), 数列最后一项当作极值(类型交替)
    v4structs: [[s_idx, e_idx, dir, valid, s_date, e_date], ...] dir: 1涨/-1跌/0横"""
    if last_close is not None and len(piv) >= 1 and piv[-1]['i'] < n - 1:
        t = 'L' if piv[-1]['t'] == 'H' else 'H'
        piv = piv + [{'i': n - 1, 't': t, 'v': round(last_close, 2)}]
    m = len(piv)
    if m < 2:
        return [0] * n, [[0, n - 1, 0]], piv, []
    # 相邻同型比较 cmp[j] = sign(P[j].v - P[prev_same(j)].v)
    prev_same = [-1] * m
    lastL = lastH = -1
    for j in range(m):
        if piv[j]['t'] == 'L':
            prev_same[j] = lastL
            lastL = j
        else:
            prev_same[j] = lastH
            lastH = j
    cmpv = [0] * m
    for j in range(2, m):
        p = prev_same[j]
        if p >= 0:
            cmpv[j] = sign(piv[j]['v'] - piv[p]['v'])
    # 分段
    structs = []  # [s_idx, e_idx, dir]
    DIR = 0
    s0 = 0
    for j in range(2, m):
        c = cmpv[j]
        if DIR == 0:
            if c != 0 and cmpv[j - 1] == c and j - 3 >= s0:
                if s0 <= j - 4:
                    structs.append([s0, j - 4, 0])
                s0 = j - 3
                DIR = c

def rps_series(closes, level_codes, n_days):
    """RPS[c][t] = close[t]/close[t-60]-1 在同级板块中的百分位 (0~100)"""
    rps = {c: [None] * n_days for c in level_codes}
    for t in range(RPS_WIN, n_days):
        vals = []
        for c in level_codes:
            cl = closes[c]
            base, cur = cl[t - RPS_WIN], cl[t]
            if base is not None and cur is not None and base > 0:
                vals.append((c, cur / base - 1))
        if len(vals) < 5:
            continue
        for j, (c, chg) in enumerate(vals):
            less = sum(1 for _, x in vals if x < chg)
            rps[c][t] = less / (len(vals) - 1) * 100
    return rps


def run_arm(dates, codes, closes, states, rps, allA_close, t0, t1, mode, names):
    """mode: A=B方案 / B=纯RPS / C=纯NType"""
    trades, open_pos = [], {}
    for t in range(t0, t1 + 1):
        for c in codes:
            st, st_prev = states[c][t], states[c][t - 1] if t > 0 else None
            r, r_prev = rps[c][t], rps[c][t - 1] if t > 0 else None
            if mode == 'A':
                sig = (st == 1 and st_prev != 1) and (r is not None and r >= RPS_TH)
            elif mode == 'B':
                sig = (r is not None and r >= RPS_TH) and (r_prev is None or r_prev < RPS_TH)
            else:
                sig = (st == 1 and st_prev != 1)
            if sig and c not in open_pos:
                open_pos[c] = t
        for c in list(open_pos):
            ts = open_pos[c]
            entry_t = ts + 1
            if entry_t > t1:
                del open_pos[c]
                continue
            if mode in ('A', 'C'):
                exit_sig = t if states[c][t] != 1 else None
            else:
                exit_sig = t if (rps[c][t] is not None and rps[c][t] < RPS_TH) else None
            if exit_sig is not None and exit_sig > ts:
                exit_t = min(exit_sig + 1, t1)
                entry_p, exit_p = closes[c][entry_t], closes[c][exit_t]
                if entry_p and exit_p and entry_p > 0:
                    ret = exit_p / entry_p - 1
                    b0, b1 = allA_close[entry_t], allA_close[exit_t]
                    excess = ret - (b1 / b0 - 1) if (b0 and b1 and b0 > 0) else None
                    trades.append(dict(code=c, name=names.get(c, c), sig_t=ts,
                                       entry_t=entry_t, exit_t=exit_t, ret=ret,
                                       excess=excess, hold=exit_t - entry_t))
                del open_pos[c]
    return trades


def run_random(dates, codes, closes, allA_close, t0, t1, n, names):
    """随机臂: n 个随机(板块, 信号日) → T+1入场 → 固定持有20日"""
    random.seed(RAND_SEED)
    trades = []
    tries = 0
    while len(trades) < n and tries < n * 20:
        tries += 1
        c = random.choice(codes)
        ts = random.randint(t0, t1)
        entry_t, exit_t = ts + 1, ts + 1 + RAND_HOLD
        if entry_t > t1 or exit_t > t1:
            continue
        entry_p, exit_p = closes[c][entry_t], closes[c][exit_t]
        if entry_p and exit_p and entry_p > 0:
            ret = exit_p / entry_p - 1
            b0, b1 = allA_close[entry_t], allA_close[exit_t]
            excess = ret - (b1 / b0 - 1) if (b0 and b1 and b0 > 0) else None
            trades.append(dict(code=c, name=names.get(c, c), sig_t=ts,
                               entry_t=entry_t, exit_t=exit_t, ret=ret,
                               excess=excess, hold=exit_t - entry_t))
    return trades


def stats(trades, label, allA_close):
    if not trades:

    rets = np.array([x['ret'] for x in trades])

    holds = np.array([x['hold'] for x in trades])
    return {
        'label': label,
        'n': len(trades),
        'mean_ret': round(float(rets.mean()) * 100, 2),
        'median_ret': round(float(np.median(rets)) * 100, 2),

def causal_states(closes, W):

def load_sw():
    """sw_index_daily → dates + closes (按层级) + 全A基准(000985中证全指或等权)"""
    c = sqlite3.connect(DB)
    lvmap = dict(c.execute("SELECT index_code, level FROM sw_classify").fetchall())
    nmap = dict(c.execute("SELECT index_code, industry_name FROM sw_classify").fetchall())
    rows = c.execute("SELECT index_code, trade_date, close FROM sw_index_daily ORDER BY trade_date").fetchall()
    dates = sorted({r[1] for r in rows})
    D = {d: i for i, d in enumerate(dates)}
    levels = {'L1': [], 'L2': [], 'L3': []}
    closes = {}
    by = {}
    for code, d, cl in rows:
        by.setdefault(code, {})[d] = cl
    for code in by:
        lv = lvmap.get(code)
        if lv not in levels:
            continue
        levels[lv].append(code)
        closes[code] = [by[code].get(d) for d in dates]
    # 基准: 等权全板块均值(每天有数据的板块均值涨跌累积) 近似"市场"
    bench = []
    acc = 1.0
    for t in range(len(dates)):
        vals = [closes[c][t] / closes[c][t - 1] for c in closes
                if closes[c][t] and closes[c][t - 1] and closes[c][t - 1] > 0]
        if vals and t > 0:
            acc *= float(np.mean(vals))
        bench.append(acc)
    return dates, D, levels, closes, nmap, bench

        codes = levels[lv]
        st = {}
        for code in codes:
            st[code] = bpv.causal_states(closes[code], V4W)
        rps = {}
        for t in range(RPS_WIN, n):
            vals = []
            for code in codes:
                b, cur = closes[code][t - RPS_WIN], closes[code][t]
                if b and cur and b > 0:
                    vals.append((code, cur / b - 1))
            if len(vals) < 5:
                continue
            for code, chg in vals:
                less = sum(1 for _, x in vals if x < chg)
                rps.setdefault(code, {})[t] = less / (len(vals) - 1) * 100

        trades, open_pos = [], {}
        for t in range(tA0, tA1 + 1):
            for code in codes:
                s, s_prev = st[code][t], st[code][t - 1] if t > 0 else None
                r = rps.get(code, {}).get(t)
                if (s == 1 and s_prev != 1) and (r is not None and r >= RPS_TH) and code not in open_pos:
                    open_pos[code] = t
            for code in list(open_pos):
                ts = open_pos[code]
                entry_t = ts + 1
                if entry_t > tA1:
                    del open_pos[code]
                    continue
                if st[code][t] != 1:
                    exit_t = min(t + 1, tA1)
                    ep, xp = closes[code][entry_t], closes[code][exit_t]
                    if ep and xp and ep > 0:

                        trades.append(dict(code=code, name=nmap.get(code, code),
                                           sig_t=ts, entry_t=entry_t, exit_t=exit_t,
                                           entry_p=round(ep, 2), exit_p=round(xp, 2),
                                           ret=round(ret * 100, 2), hold=exit_t - entry_t,
                                           rps=round(rps.get(code, {}).get(ts, -1), 1)))
                    del open_pos[code]
        all_trades[lv] = trades
        rets = np.array([x['ret'] for x in trades])
        report['by_level'][lv] = {
            'n': len(trades),
            'mean_ret': round(float(rets.mean()), 2) if len(rets) else None,
            'median_ret': round(float(np.median(rets)), 2) if len(rets) else None,
            'win_rate': round(float((rets > 0).mean()) * 100, 1) if len(rets) else None,
            'open': len(open_pos),
        }
        s = report['by_level'][lv]
        print(f"{lv}: n={s['n']} 均收{s['mean_ret']}% 中位{s['median_ret']}% 胜率{s['win_rate']}% 未平仓{s['open']}")

    c.execute("""CREATE TABLE IF NOT EXISTS sw_index_daily (
        index_code TEXT, trade_date TEXT, open REAL, high REAL, low REAL, close REAL,
        PRIMARY KEY (index_code, trade_date))""")
    done = set(r[0] for r in c.execute("SELECT DISTINCT index_code FROM sw_index_daily").fetchall())
    codes = [r[0] for r in c.execute(
        "SELECT index_code FROM sw_classify ORDER BY level, industry_code").fetchall()]
    codes = [x for x in codes if x not in done]
    print(f'待拉 {len(codes)} 个指数 (已跳过 {len(done)}), {START}~{END}', flush=True)
    ok = fail = 0
    nodata = []
    for i, code in enumerate(codes):
        df = None
        for attempt in range(3):
            try:
                df = pro.index_daily(ts_code=code, start_date=START, end_date=END)
                break
            except Exception as e:
                err = str(e)[:80]
                if 'limit' in err.lower() or '频率' in err or '超限' in err:
                    print(f'  限频, 等65s ... ({code}, attempt {attempt+1})', flush=True)
                    time.sleep(65)
                else:
                    print(f'  {code} FAIL: {err}', flush=True)
                    time.sleep(1)
                    break
        if df is None or df.empty:
            nodata.append(code)
            fail += 1
        else:
            rows = [(code, r.trade_date, float(r.open), float(r.high), float(r.low), float(r.close))
                    for r in df.itertuples()]
            c.executemany("INSERT OR REPLACE INTO sw_index_daily VALUES (?,?,?,?,?,?)", rows)
            c.commit()
            ok += 1
        time.sleep(0.55)  # ~110次/分钟, 低于 200 上限
        if (i + 1) % 40 == 0:
            print(f'  {i+1}/{len(codes)} ok={ok} fail={fail}', flush=True)
    n = c.execute("SELECT COUNT(*) FROM sw_index_daily").fetchone()[0]
    nc = c.execute("SELECT COUNT(DISTINCT index_code) FROM sw_index_daily").fetchone()[0]
    print(f'完成: ok={ok} fail={fail}, 总行 {n}, 指数 {nc}', flush=True)
